Each word but the last is padded with null bytes to a full row, like the char[8] fields of a C struct, so words are never split across rows, and an empty word is a row of null bytes. Text is dumped as UTF-8. It is fully rendered here, custom.js only adds the scramble and the byte <-> character pairing. */ -}} {{- $cols := 8 }} {{- $label := slice }} {{- $hex := "" }} {{- range $i, $word := . }} {{- with $word }}{{ $label = $label | append . }}{{ end }} {{- /* "%x" on a string prints the hex of its UTF-8 bytes */ -}} {{- $h := printf "%x" $word }} {{- if lt $i (sub (len $) 1) }} {{- $pad := $cols }} {{- with $word }}{{ $pad = mod (sub $cols (mod (div (len $h) 2) $cols)) $cols }}{{ end }} {{- $h = print $h (strings.Repeat $pad "00") }} {{- end }} {{- $hex = print $hex $h }} {{- end }} {{- $bytes := findRE `..` $hex }}
